on_actions = {
	
	on_startup = {
		effect = {
			1037 = { 
				add_dynamic_modifier = { modifier = RAJ_naxalite_warloads }
			}
			431 = { 
				add_dynamic_modifier = { modifier = RAJ_naxalite_warloads }
			}
			424 = { 
				add_dynamic_modifier = { modifier = RAJ_naxalite_warloads }
			}
			427 = { 
				add_dynamic_modifier = { modifier = RAJ_naxalite_warloads }
			}
		}
	}

	on_nuke_drop = {
		effect = {
			if = {
				limit = {
					FROM = {
						is_core_of = RAJ
					}
					has_global_flag = RAJ_nuke_1
					NOT = {
						has_global_flag = RAJ_nuke_2
					}
				}
				set_global_flag = RAJ_nuke_2
			}
			if = {
				limit = {
					FROM = {
						is_core_of = RAJ
					}
					NOT = {
						has_global_flag = RAJ_nuke_1
					}
				}
				set_global_flag = RAJ_nuke_1
			}
			
		}
	}
	
	on_ruling_party_change = {
		effect = {
			if = {
				limit = {
					original_tag = RAJ
					NOT = { has_government = reactionary	}
					has_power_balance = {
						id = RAJ_power_balance
					}
				}
				remove_power_balance = {
					id = RAJ_power_balance
				}
			}
		}
		effect = {
			if = {
				limit = {
					original_tag = RAJ
					has_government = reactionary
					NOT = { has_power_balance = {
						id = RAJ_power_balance
					} }
				}
				set_power_balance = {
					id = RAJ_power_balance
					left_side = RAJ_INS_side
					right_side = RAJ_BJP_side
				}
			}
		}
	}
	
}
